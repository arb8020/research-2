# Extra rows on a per_token (full attention) group, Kmax = 3. A commit asks for extra rows on
# the request's next decode-shaped entry (next=, default Kmax here); the plan gives that entry
# 1 history row + 3 extra rows whose input_ids are no_token (`-`): the caller fills them (a
# drafter's drafts). kept=n says tail rows 0..n-1 were valid.
#
# Rows exist exactly for positions < seq_start + 1 + 3: no reach term past the plan's own rows
# (the drafter's noise block IS the extra rows, docs/design-core-option2.md section 0).
config max_seq_len=64 max_batch_tokens=64 max_batch_reqs=4 block_size=4 groups=full num_blocks=16 max_extra=3 reserve_ratio=0 order=running_first eos=99
submit A prompt=8 max_new=16
# A prefill never carries extra rows, whatever the last commit asked (section 8 item 1).
step
  expect batch A:8
  expect table A g=0 [1,2]
  expect free 13
plan kept=A:3
  expect batch A:1+3
  expect rows A [1@8,-@9,-@10,-@11]
  expect table A g=0 [1,2,3]
commit
# kept=3: extra rows 1 and 2 were valid (tokens 1, 2), then token 3. n_computed 8 -> 11.
  expect n_computed A 11
  expect event n_extra_rows=3 n_extra_kept=2
  expect free 12
# kept=1: no extra row kept, exactly a plain decode. Row 3 (positions 12..15) was allocated
# for the extra rows and stays as the request's headroom.
plan kept=A:1
  expect rows A [3@11,-@12,-@13,-@14]
  expect table A g=0 [1,2,3,4]
commit
  expect n_computed A 12
  expect free 11
# kept=4 = 1 + Kmax: every extra row valid; nothing new is allocated (rows through 15 exist).
plan kept=A:4 next=A:0
  expect rows A [1@12,-@13,-@14,-@15]
commit
  expect n_computed A 16
  expect free 11
# next=0: the following entry is a plain decode.
step
  expect batch A:1
  expect table A g=0 [1,2,3,4,5]
  expect free 10

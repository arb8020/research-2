# Cancel a request that only the ahead plan names: B is admitted by N+1 while N (A's decode) is in
# flight. cancel() leaves B's blocks alone (planned_ahead), so the commit that finally handles
# B's entry, commit(N+1), frees them before it releases the slot.
config max_batch_tokens=16 max_batch_reqs=4 block_size=4 num_blocks=16 reserve_ratio=0 order=running_first eos=99 overlap=1
submit A prompt=4 max_new=8
step tokens=A:7
plan tokens=A:8
  expect batch A:1                  # N: A's decode
submit B prompt=6 max_new=4
plan tokens=A:9,B:50
  expect ahead_batch A:1 B:6        # N+1 admits B
cancel B
commit                              # commits N: B is not in it
  expect polled A [7,8]
commit                              # commits N+1: B cancelled; its blocks go back
  expect running A
  expect polled A [9]
  expect free 13

# Overlap: the plan built ahead (N+1) must not re-plan a prefill chunk that is still in flight.
# A's whole 10-token prompt goes out in N, and that chunk emits A's first token. N+1 plans from
# committed + in-flight = 10: a decode of N's (not yet sampled) token, reserved like any decode,
# at position 10. It used to re-plan A:10 at positions 0..9, and the second commit panicked in
# request.advance (NOTES-rung0.md, "overlap bug, NOT fixed").
config max_batch_tokens=16 max_batch_reqs=4 block_size=4 num_blocks=16 reserve_ratio=0 order=running_first eos=99 overlap=1
submit A prompt=10 max_new=4
plan tokens=A:50
  expect batch A:10                 # N: the whole prompt, emits A's first token
  expect rows A [1000@0,1001@1,1002@2,1003@3,1004@4,1005@5,1006@6,1007@7,1008@8,1009@9]
plan tokens=A:51
  expect ahead_batch A:1            # N+1: the decode after N's chunk, not the chunk again
  expect ahead_rows A [-@10]        # placeholder for N's token, at committed 0 + in flight 10
  expect reserved A true
commit                              # commits N: the placeholder becomes 50
  expect n_computed A 10
  expect polled A [50]
commit                              # commits N+1
  expect n_computed A 11
  expect polled A [51]

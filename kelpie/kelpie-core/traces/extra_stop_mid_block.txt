# A commit keeps tail rows only up to the first stop token and within max_new (T5). A proposer
# may draft PAST an eos; none of that may enter the history, advance n_computed, or pick the live
# state. A per_seq group shows which write unit becomes live.
config max_seq_len=64 max_batch_tokens=64 max_batch_reqs=4 block_size=4 groups=full,state8 num_blocks=40 max_extra=3 reserve_ratio=0 order=running_first eos=99
submit A prompt=4 base=1000 max_new=16
submit B prompt=4 base=2000 max_new=2
submit C prompt=4 base=3000 max_new=16
step
  expect batch A:4 B:4 C:4
  expect free 32
# A commits [98, 99, 100]: 99 is eos, so it keeps 2 and finishes. B has room for 1 token of the
# 4 it commits (max_new 2, 1 generated): it keeps 1 and finishes. C keeps 3: live := write[2].
plan tokens=A:98 kept=A:3,B:4,C:3
  expect batch A:1+3 B:1+3 C:1+3
  expect idx A g=1 start=4 read=3 write=[3,9,10,11] ckpt=-
  expect idx B g=1 start=4 read=5 write=[5,13,14,15] ckpt=-
  expect idx C g=1 start=4 read=7 write=[7,17,18,19] ckpt=-
commit
  expect finished A reason=finished_eos
  expect finished B reason=finished_length
  expect polled A [1,98,99]
  expect polled B [1,1]
  expect polled C [1,1,2,3]
  expect n_computed C 7
  expect units C g=1 live=18 ring=[]
  expect free 35
# C's history row is position 7: its state after row 0 completes 8 -- a checkpoint on the
# history row, with the extra rows after it. (F row 20 is allocated first: rows through 10.)
plan kept=C:1
  expect rows C [3@7,-@8,-@9,-@10]
  expect idx C g=1 start=7 read=18 write=[18,21,22,23] ckpt=24@row0
commit
  expect units C g=1 live=18 ring=[24@8]

# Same submissions as running_first, but admission takes the budget first: C prefills whole,
# A gets one decode, B's remaining 2 prompt tokens cannot be block aligned into 1 token so B waits a step.
config max_batch_tokens=8 max_batch_reqs=4 block_size=4 num_blocks=16 reserve_ratio=0 order=waiting_first eos=99
submit A prompt=4 max_new=4
submit B prompt=6 max_new=2
step
  expect batch A:4 B:4
  expect free 13
submit C prompt=6 max_new=2
step
  expect batch A:1 C:6
  expect free 10
  expect running A B C
step
  expect batch A:1 C:1 B:2
  expect free 11               # C hits max_new=2 and frees its 2 blocks; B takes 1
  expect running A B
expect finished C reason=finished_length

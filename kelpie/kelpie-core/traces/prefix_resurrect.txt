# A freed block keeps its hash on the free list and is resurrected by a later hit; a block popped by allocate
# forgets its hash. num_blocks=8 -> 7 usable, block ids 1..7 handed out LRU. Every request runs two steps
# (prefill, one decode) so its block table can be inspected after the prefill.
config max_batch_tokens=16 max_batch_reqs=4 block_size=4 num_blocks=8 prefix_cache=1
submit A prompt=8 max_new=2
step
  expect batch A:8
  expect blocks A 1 2
step
  expect batch A:1              # token 8 lands in block 3
expect finished A reason=finished_length
expect polled A [1,1]
expect free 7                   # free list: 4 5 6 7 3 2 1 (A released its tail first); blocks 1 and 2 stay hashed
submit B prompt=8 max_new=2
step
  expect batch B:4              # block 1 hits (tokens 0..3); tokens 4..7 contain the last token and are recomputed
  expect blocks B 1 4           # 1 resurrected from the middle of the free list, 4 popped from the head
  expect prefix queries=1 hits=1 tokens=4
step
expect finished B reason=finished_length
expect polled B [1,1]
expect free 7                   # free list: 6 7 3 2 5 4 1. B's block 4 duplicates block 2's content; the table keeps block 2
submit C prompt=9 base=5000 max_new=2
step
  expect batch C:9
  expect blocks C 6 7 3         # unrelated prompt, no hits; pops 6 7 3
  expect prefix queries=1 hits=0 tokens=0
step
expect finished C reason=finished_length
expect polled C [1,1]
submit D prompt=9 max_new=2
step
  expect batch D:1              # 9 tokens: blocks 1 and 2 (tokens 0..7) both hit, only token 8 is computed
  expect blocks D 1 2 5         # 1 and 2 resurrected; 5 is the free-list head (2 5 4 1 3 7 6 after C released 3 7 6)
  expect prefix queries=1 hits=1 tokens=8

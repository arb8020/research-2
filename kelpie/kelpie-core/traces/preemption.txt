# Four usable blocks (num_blocks=5; block 0 is null). A and B each hold 2 blocks by step 2. When A needs a third block (n_computed 8 -> 9)
# the youngest runner B is evicted to the front of waiting; nothing is admitted in a step that preempted.
# B keeps its 5 generated tokens and re-prefills all 9 when A finishes.
config max_batch_tokens=16 max_batch_reqs=4 block_size=4 num_blocks=5 reserve_ratio=0 order=running_first eos=99
submit A prompt=4 max_new=8
submit B prompt=4 max_new=8
step
  expect batch A:4 B:4
  expect free 2
  expect event n_admitted=2 n_preempted=0 n_running=2 kv_free=2 kv_used=2
step
  expect batch A:1 B:1
  expect free 0
step
step
step
  expect batch A:1 B:1
  expect free 0
  expect running A B
step
  expect batch A:1
  expect free 1
  expect running A
  expect waiting B
  expect event n_admitted=0 n_preempted=1 n_finished=0 n_running=1 n_waiting=1
step
  expect batch A:1
  expect free 1
  expect waiting B
step
  expect batch A:1
  expect free 4
  expect running
  expect waiting B
  expect event n_finished=1 n_running=1 n_decode_reqs=1
expect finished A reason=finished_length
expect polled A [1,1,1,1,1,1,1,1]
step tokens=B:2

  expect free 1
  expect running B
  expect waiting
expect polled B [1,1,1,1,1,2]
